(function($) {
  function setupCustomMenuBehavior($menu) {
    $menu.find('.menu-item-has-children').each(function () {
      var $item = $(this);
      var $link = $item.find('> .fl-has-submenu-container > a');
      var $toggle = $item.find('> .fl-has-submenu-container > .fl-menu-toggle');

      // Remove all default click handlers
      $item.off('click');
      $link.off('click');
      $toggle.off('click');

        e.stopPropagation(); // allow the link, just prevent bubbling

        // Toggle current

          $item.removeClass('fl-active');
